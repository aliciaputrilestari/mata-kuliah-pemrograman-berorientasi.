LATIHAN 1: CONNECTION FACTORY

Membuat koneksi database itu *lambat dan berat. Daripada buat koneksi baru setiap kali dibutuhkan, kita simpan koneksi yang sudah ada di **pool*. Kalau butuh koneksi, cek pool dulu — kalau ada yang tersedia langsung pakai, kalau belum baru buat baru. Ini menghemat waktu dan resource komputer.

// Nama  : Alicia Putri Lestari
// Nim   : 251410011
// Kelas : SI2A

import 'dart:collection';

// Latihan 1: Connection Factory dengan Pooling
class DatabaseConnection {
  final String connectionString;
  bool _isUsed = false;

  DatabaseConnection(this.connectionString);

  void connect() => print('✅ Terhubung ke: $connectionString');
  void disconnect() => print('❌ Terputus dari: $connectionString');

  bool get isUsed => _isUsed;
  set isUsed(bool value) => _isUsed = value;
}

class ConnectionFactory {
  final List _pool = [];
  final String _defaultConnStr;
  final int _maxPoolSize;

  ConnectionFactory(this._defaultConnStr, [this._maxPoolSize = 5]);

  DatabaseConnection getConnection() {
    // Cari koneksi yang tidak terpakai
    for (var conn in _pool) {
      if (!conn.isUsed) {
        conn.isUsed = true;
        return conn;
      }
    }

    // Buat baru jika belum penuh
    if (_pool.length < _maxPoolSize) {
      final newConn = DatabaseConnection(_defaultConnStr);
      newConn.isUsed = true;
      _pool.add(newConn);
      return newConn;
    }

    throw Exception('⚠️ Koneksi habis! Maksimal $_maxPoolSize koneksi.');
  }

  void releaseConnection(DatabaseConnection conn) {
    conn.isUsed = false;
    print('🔓 Koneksi dikembalikan ke pool');
  }

  int get activeConnections => _pool.where((c) => c.isUsed).length;
}

void main() {
  print('=== UJI COBA CONNECTION FACTORY ===');
  final factory = ConnectionFactory('mysql://localhost:3306/db_saya', 3);

  try {
    final conn1 = factory.getConnection();
    conn1.connect();
    print('Koneksi aktif: ${factory.activeConnections}');

    final conn2 = factory.getConnection();
    conn2.connect();
    print('Koneksi aktif: ${factory.activeConnections}');

    // Lepas koneksi pertama
    factory.releaseConnection(conn1);
    print('Koneksi aktif setelah lepas: ${factory.activeConnections}');

    // Ambil lagi, akan pakai yang lama
    final conn3 = factory.getConnection();
    conn3.connect();
    print('Koneksi aktif: ${factory.activeConnections}');

  } catch (e) {
    print(e);
  }
}

LATIHAN 2: NOTIFICATION FACTORY

Ketika aplikasi perlu kirim notifikasi, kita tidak selalu tahu harus pakai Email, SMS, atau Push. Factory ini bertugas memilihkan class yang tepat berdasarkan input platform. Kamu cukup tulis "email" atau "sms", factory yang urus pemilihan class-nya. Kalau mau tambah platform baru, tinggal daftar di _platforms tanpa ubah logika lain.

// Nama  : Alicia Putri Lestari
// Nim   : 251410011
// Kelas : SI2A

// Latihan 2: Notification Factory (Versi DartPad Fix)
abstract class Notification {
  void send(String pesan);
}

class AndroidNotification extends Notification {
  @override
  void send(String pesan) => print('📱 [Android] Notifikasi: $pesan');
}

class IOSNotification extends Notification {
  @override
  void send(String pesan) => print('🍎 [iOS] Notifikasi: $pesan');
}

class WebNotification extends Notification {
  @override
  void send(String pesan) => print('🌐 [Web] Notifikasi: $pesan');
}

class NotificationFactory {
  // Ubah: Kita gunakan parameter input untuk memilih, karena Platform tidak bisa dipakai di DartPad
  static Notification createNotification(String platform) {
    print('🔍 Platform dipilih: $platform');

    switch (platform.toLowerCase()) {
      case 'android':
        return AndroidNotification();
      case 'ios':
        return IOSNotification();
      case 'web':
      default:
        return WebNotification();
    }
  }
}

void main() {
  print('=== UJI COBA NOTIFICATION FACTORY ===');

  // Coba semua jenis platform
  final notif1 = NotificationFactory.createNotification('android');
  notif1.send('Halo Pengguna Android!');

  final notif2 = NotificationFactory.createNotification('ios');
  notif2.send('Halo Pengguna iPhone!');

  final notif3 = NotificationFactory.createNotification('web');
  notif3.send('Halo Pengguna Website!');
}

LATIHAN 3: SHAPE FACTORY

Kalau program perlu membuat banyak shape dengan tipe dan warna yang sama, tidak efisien kalau dibuat ulang terus. Factory ini menyimpan shape yang sudah dibuat di cache. Saat ada permintaan shape yang sama, langsung ambil dari cache. Hasilnya objek yang dikembalikan benar-benar objek yang sama di memori (s1 is s3 → True).

// Nama  : Alicia Putri Lestari
// Nim   : 251410011
// Kelas : SI2A

import 'dart:collection';

// Latihan 3: Shape Factory dengan Cache
abstract class Shape {
  void draw();
}

class Circle extends Shape {
  final double radius;
  Circle(this.radius);

  @override
  bool operator ==(Object other) =>
      identical(this, other) || other is Circle && radius == other.radius;

  @override
  int get hashCode => radius.hashCode;

  @override
  void draw() => print('🔵 Lingkaran | Jari-jari: $radius');
}

class Rectangle extends Shape {
  final double lebar, tinggi;
  Rectangle(this.lebar, this.tinggi);

  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
      other is Rectangle && lebar == other.lebar && tinggi == other.tinggi;

  @override
  int get hashCode => Object.hash(lebar, tinggi);

  @override
  void draw() => print('🟦 Persegi Panjang | ${lebar} x $tinggi');
}

class ShapeFactory {
  static final HashMap _cache = HashMap();

  static Shape getShape(String tipe, {double? radius, double? lebar, double? tinggi}) {
    String key;
    Shape? shape;

    if (tipe == 'lingkaran' && radius != null) {
      key = 'lingkaran-$radius';
      shape = _cache[key];
      if (shape == null) {
        print('✨ Membuat objek baru: $key');
        shape = Circle(radius);
        _cache[key] = shape;
      } else {
        print('📤 Ambil dari cache: $key');
      }
    } else if (tipe == 'persegi' && lebar != null && tinggi != null) {
      key = 'persegi-$lebar-$tinggi';
      shape = _cache[key];
      if (shape == null) {
        print('✨ Membuat objek baru: $key');
        shape = Rectangle(lebar, tinggi);
        _cache[key] = shape;
      } else {
        print('📤 Ambil dari cache: $key');
      }
    } else {
      throw Exception('⚠️ Parameter tidak lengkap atau tipe tidak dikenal!');
    }
    return shape;
  }

  static int get jumlahDiCache => _cache.length;
}

void main() {
  print('=== UJI COBA SHAPE FACTORY + CACHE ===');
  try {
    // Buat lingkaran pertama
    final l1 = ShapeFactory.getShape('lingkaran', radius: 10);
    l1.draw();

    // Sama persis, ambil dari cache
    final l2 = ShapeFactory.getShape('lingkaran', radius: 10);
    l2.draw();

    // Ukuran beda, buat baru
    final l3 = ShapeFactory.getShape('lingkaran', radius: 20);
    l3.draw();

    final p1 = ShapeFactory.getShape('persegi', lebar: 5, tinggi: 8);
    p1.draw();

    print('\n📦 Total objek tersimpan di cache: ${ShapeFactory.jumlahDiCache}');
  } catch (e) {
    print(e);
  }
}

CHALLENGE: ANIMAL FACTORY

Ini adalah factory paling lengkap yang menggabungkan semua konsep sebelumnya:

- Factory method → satu method create() untuk semua jenis hewan

- Subclass selection → input "dog" otomatis buat objek Dog, bukan Cat atau Bird

- Cache → hewan dengan nama dan species sama tidak dibuat dua kali

- Validation → sebelum dibuat, input dicek dulu (nama tidak boleh kosong, species harus valid)

// Nama  : Alicia Putri Lestari
// Nim   : 251410011
// Kelas : SI2A

import 'dart:collection';

// Challenge: Animal Factory
abstract class Animal {
  final String nama;
  final int umur;
  Animal(this.nama, this.umur);
  void bersuara();

  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
      other is Animal && nama == other.nama && umur == other.umur;

  @override
  int get hashCode => Object.hash(nama, umur);
}

class Kucing extends Animal {
  Kucing(String nama, int umur) : super(nama, umur);
  @override
  void bersuara() => print('🐈 $nama: Meong meong!');
}

class Anjing extends Animal {
  Anjing(String nama, int umur) : super(nama, umur);
  @override
  void bersuara() => print('🐕 $nama: Guk guk!');
}

class Burung extends Animal {
  Burung(String nama, int umur) : super(nama, umur);
  @override
  void bersuara() => print('🐦 $nama: Ciuwit ciuwit!');
}

class AnimalFactory {
  static final HashMap _cache = HashMap();

  static Animal createAnimal(String jenis, String nama, int umur) {
    // 1. Validasi
    if (nama.trim().isEmpty) throw Exception('⚠️ Nama tidak boleh kosong!');
    if (umur <= 0 || umur > 100) throw Exception('⚠️ Umur harus 1 - 100 tahun!');

    // Kunci cache
    final key = '${jenis.toLowerCase()}-$nama-$umur';

    // 2. Cek Cache
    if (_cache.containsKey(key)) {
      print('📤 Ambil dari cache: $key');
      return _cache[key]!;
    }

    // 3. Pilih Subclass
    Animal hewan;
    switch (jenis.toLowerCase()) {
      case 'kucing':
        hewan = Kucing(nama, umur);
        break;
      case 'anjing':
        hewan = Anjing(nama, umur);
        break;
      case 'burung':
        hewan = Burung(nama, umur);
        break;
      default:
        throw Exception('⚠️ Jenis "$jenis" tidak tersedia! (kucing/anjing/burung)');
    }

    // 4. Simpan ke Cache
    _cache[key] = hewan;
    print('✨ Objek baru dibuat: $key');
    return hewan;
  }

  static int get totalCache => _cache.length;
}

void main() {
  print('=== UJI COBA ANIMAL FACTORY ===');
  try {
    final a1 = AnimalFactory.createAnimal('kucing', 'Mochi', 3);
    a1.bersuara();

    final a2 = AnimalFactory.createAnimal('kucing', 'Mochi', 3); // Sama, ambil cache
    a2.bersuara();

    final a3 = AnimalFactory.createAnimal('anjing', 'Bobby', 5);
    a3.bersuara();

    final a4 = AnimalFactory.createAnimal('burung', 'Kutilang', 1);
    a4.bersuara();

    print('\n📦 Total hewan unik tersimpan: ${AnimalFactory.totalCache}');

    // Contoh error validasi (buka komentar untuk coba)
    // AnimalFactory.createAnimal('sapi', 'Sapi Kecil', 2); // Jenis salah
    // AnimalFactory.createAnimal('kucing', '', 2); // Nama kosong
    // AnimalFactory.createAnimal('anjing', 'Blacky', -5); // Umur salah

  } catch (e) {
    print(e);
  }
}
